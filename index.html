begin; -- 用登入信箱的 @ 前帳號辨識；不更動密碼或登入信箱。 update public.profiles p set display_name=case lower(split_part(u.email,'@',1)) when 'kevindd100' then '鬍子教練' when 'weiihuang49' then '威力教練' end from auth.users u where p.id=u.id and lower(split_part(u.email,'@',1)) in ('kevindd100','weiihuang49'); create table if not exists public.coach_update_logs ( id bigint generated always as identity primary key, created_at timestamptz not null default clock_timestamp(), actor_id uuid, actor_name text not null, entity text not null check(entity in ('meets','performances')), entity_id uuid not null, action text not null check(action in ('INSERT','UPDATE','DELETE','ARCHIVE','RESTORE')), before_data jsonb, after_data jsonb ); create index if not exists coach_update_logs_recent on public.coach_update_logs(id desc); alter table public.coach_update_logs enable row level security; revoke all on public.coach_update_logs from anon,authenticated; grant select on public.coach_update_logs to authenticated; drop policy if exists coach_logs_read on public.coach_update_logs; create policy coach_logs_read on public.coach_update_logs for select to authenticated using(public.is_coach()); create or replace function public.log_coach_update() returns trigger language plpgsql security definer set search_path=public as $$ declare b jsonb; a jsonb; actor uuid:=auth.uid(); label text; verb text:=TG_OP; begin if TG_OP <> 'INSERT' then b:=to_jsonb(OLD); end if; if TG_OP <> 'DELETE' then a:=to_jsonb(NEW); end if; if TG_OP='UPDATE' and a=b then return NEW; end if; select nullif(trim(p.display_name),'') into label from public.profiles p where p.id=actor; label:=coalesce(label,case when actor is null then '系統／SQL 更新' else '未命名教練' end); if TG_TABLE_NAME='meets' and TG_OP='UPDATE' then if b->>'active'='true' and a->>'active'='false' then verb:='ARCHIVE'; elsif b->>'active'='false' and a->>'active'='true' then verb:='RESTORE'; end if; end if; insert into public.coach_update_logs(actor_id,actor_name,entity,entity_id,action,before_data,after_data) values(actor,label,TG_TABLE_NAME,coalesce((a->>'id')::uuid,(b->>'id')::uuid),verb,b,a); if TG_OP='DELETE' then return OLD; end if; return NEW; end $$; revoke all on function public.log_coach_update() from public,anon,authenticated; drop trigger if exists coach_audit_meets on public.meets; create trigger coach_audit_meets after insert or update or delete on public.meets for each row execute function public.log_coach_update(); drop trigger if exists coach_audit_performances on public.performances; create trigger coach_audit_performances after insert or update or delete on public.performances for each row execute function public.log_coach_update(); commit; -- 以下應看到兩位教練的新名稱。 select p.display_name,split_part(u.email,'@',1) as account from public.profiles p join auth.users u on u.id=p.id where lower(split_part(u.email,'@',1)) in ('kevindd100','weiihuang49');
